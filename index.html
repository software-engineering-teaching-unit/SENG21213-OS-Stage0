About This Assignment

Overview

What You Will Build

This assignment asks you to build a minimal but functional operating system in C and x86-32 NASM assembly. Starting from nothing but a BIOS-loaded 512-byte sector, you will add subsystems stage by stage — a scheduler, concurrency primitives, a memory manager, and finally a file system — mirroring the sequence of OS lectures L07 through L12.

Every stage produces a system that boots in QEMU and passes shell-level tests. Each stage builds on the previous one. Read each stage article fully before writing any code.

Textbook alignment

All concepts are in Stallings, Computer Organization and Architecture, 10th edition. The lecture notes L07–L12 cover the theory; this guide is the implementation companion.

Frequently Asked Questions
  • QEMU shows a black screen. The kernel panicked before vga_init() ran, or the boot sector didn't load. Run make debug + make gdb and step through boot.asm to find where it stops.
  • "No rule to make target". Run make clean then make. Check that all source files listed in the Makefile's KERN_SRCS variable exist.
  • GDB says "Remote connection closed". QEMU exited (likely a triple fault). Add a breakpoint earlier — try break kernel_main before continue.
  • Characters appear garbled in QEMU. The attribute byte and character byte are swapped in vga_cell(). Check the bit-shift direction: (attr << 8) | (uint8_t)c.
  • Context switch causes a reboot. The stack offsets in switch.asm are wrong. After PUSHAD (8 × 4 = 32 bytes) + return address (4 bytes) the arguments are at [esp+36] and [esp+40].
  • Stage 3: pmm_alloc_frame() always returns 0. pmm_init() didn't run, or the E820 map wasn't written to 0x8000 by the bootloader. Add a vga_printf in pmm_init() to print the entry count.
  • Can I use a different language? No. The assignment requires C99 and NASM x86-32 assembly as specified.

Boot, VGA & Shell

Lectures 07–08 · Stallings Ch. 2, 11 · Starter kit provided

Stage 0

What you need to understand first

Read L07 §4 (x86 boot sequence, GDT, Protected Mode) and L08 §1 (VGA text mode, I/O ports) before writing any code. The starter kit implements everything in Stage 0 — your job is to understand it thoroughly, get it running, and implement at least one extension.

① The Boot Sequence — 8 ordered steps
#ActionWhy it matters
1CLI — disable interruptsPrevent IRQs firing while GDT is being loaded
2Set up stack at 0x7C00C code and CALL need a stack
3INT 0x13 AH=0x42 — extended readLoad kernel from disk; last BIOS call before mode switch
4LGDT [gdt_descriptor]Tell CPU where the 3-entry GDT lives
5OR CR0, 1 — set PE bitProtection Enable — enters Protected Mode
6JMP 0x08:pm_entry (FAR JUMP)Flushes the prefetch queue; reloads CS from GDT
7Set DS/ES/SS = 0x10Load data segment selector for 32-bit flat mode
8JMP kernel_mainHand off to C — never returns to boot.asm

Why the FAR JUMP at step 6?

After writing to CR0 the CPU is in Protected Mode but the code-segment cache still holds Real Mode attributes. The FAR JUMP forces a CS reload from the GDT, completing the transition. Without it the next instruction decodes with wrong operand sizes.

② GDT — 3 descriptors, 24 bytes total
; Entry 0: null (required — selector 0x00)
gdt_null: dd 0, 0

; Entry 1: code segment (selector 0x08) — base=0, limit=4GB, ring 0, 32-bit
gdt_code:
    dw 0xFFFF   ; Limit 15:0
    dw 0x0000   ; Base  15:0
    db 0x00     ; Base  23:16
    db 0x9A     ; P=1 DPL=0 S=1 Type=Code R/X
    db 0xCF     ; G=1(4KB) DB=1(32-bit) Limit 19:16
    db 0x00     ; Base  31:24

; Entry 2: data segment (selector 0x10) — same base/limit, type=Data R/W
gdt_data:
    dw 0xFFFF  dw 0x0000  db 0x00
    db 0x92     ; P=1 DPL=0 S=1 Type=Data R/W
    db 0xCF    db 0x00
③ VGA Driver — 0xB8000 text buffer
#define VGA_BUFFER ((volatile uint16_t *)0xB8000)
/* Each cell = 2 bytes: high=attribute (bg<<4|fg), low=ASCII */
static inline uint16_t vga_cell(char c, uint8_t attr) {
    return (uint16_t)((attr << 8) | (uint8_t)c);
}
/* Hardware cursor via CRT ports 0x3D4/0x3D5 */
void vga_update_hw_cursor(int row, int col) {
    uint16_t pos = (uint16_t)(row * 80 + col);
    outb(0x3D4, 0x0F); outb(0x3D5, pos & 0xFF);
    outb(0x3D4, 0x0E); outb(0x3D5, pos >> 8);
}

Process Table & Round-Robin Scheduler

Lecture 09 · Stallings Ch. 3, 9 · New: process.h/c, scheduler.h/c, idt.h/c, switch.asm

Stage 1
① PCB struct — Process Control Block
#define MAX_PROCS  16
#define STACK_SIZE 4096   /* 4 KB kernel stack per process */

typedef enum {
    PROC_UNUSED=0, PROC_READY, PROC_RUNNING, PROC_BLOCKED, PROC_ZOMBIE
} proc_state_t;

typedef struct {
    uint32_t      pid;
    proc_state_t  state;
    uint32_t      esp;          /* saved ESP — set on context switch */
    uint32_t      stack_base;
    void        (*entry)(void);
    char          name[32];
    uint32_t      ticks;        /* total CPU ticks consumed */
} pcb_t;
② IDT + PIC remapping + PIT at 100 Hz
/* Remap PIC: IRQ0-7 → vectors 32-39 */
outb(0x20, 0x11); outb(0xA0, 0x11);  /* ICW1 */
outb(0x21, 0x20); outb(0xA1, 0x28);  /* ICW2: offsets */
outb(0x21, 0x04); outb(0xA1, 0x02);  /* ICW3 */
outb(0x21, 0x01); outb(0xA1, 0x01);  /* ICW4: 8086 mode */
outb(0x21, 0x00); outb(0xA1, 0xFF);  /* Unmask IRQ0 only */

/* Program PIT channel 0: 100 Hz (divisor = 1193182 / 100) */
outb(0x43, 0x36);          /* Mode 3 square wave */
outb(0x40, 11931 & 0xFF); /* Divisor low byte  */
outb(0x40, 11931 >> 8);   /* Divisor high byte */
③ Context switch in NASM — PUSHAD/POPAD
; context_switch(uint32_t *old_esp, uint32_t new_esp)
; After PUSHAD (32 bytes) + ret addr (4 bytes): args at [esp+36], [esp+40]
global context_switch
context_switch:
    pushad                  ; Save EAX ECX EDX EBX ESP EBP ESI EDI
    mov  eax, [esp + 36]    ; old_esp pointer
    mov  [eax], esp         ; *old_esp = current ESP
    mov  esp, [esp + 40]    ; switch to new process stack
    popad                   ; Restore next process's registers
    ret                     ; Jump into next process's code
④ Round-Robin scheduler — called from IRQ0 handler
void scheduler_tick(void) {
    proc_table[current_proc].ticks++;
    int next = (current_proc + 1) % MAX_PROCS, searched = 0;
    while (searched++ < MAX_PROCS) {
        if (proc_table[next].state == PROC_READY) break;
        next = (next + 1) % MAX_PROCS;
    }
    if (searched > MAX_PROCS) return;
    int old = current_proc;
    proc_table[old].state  = PROC_READY;
    proc_table[next].state = PROC_RUNNING;
    current_proc = next;
    outb(0x20, 0x20);         /* EOI — send BEFORE switch! */
    context_switch(&proc_table[old].esp, proc_table[next].esp);
}

Send EOI before context_switch()

After context_switch() returns you are in a different process. If you send the PIC EOI after the switch, the original process may never run again to send it — timer ticks will stop.

Extension Ideas — Stage 1 (bonus marks)

Threads, Mutex & Semaphore

Lecture 10 · Stallings Ch. 4–5 · New: thread.h/c, mutex.h/c, semaphore.h/c

Stage 2
① Race condition — myglobal demo

Run this without a mutex first. Observe that myglobal is never exactly 200000. Then add a mutex and observe it is always exactly 200000. This is the empirical proof of Dijkstra's six requirements from L10.

static volatile int myglobal = 0;
void thread_a(void) {
    for (int i = 0; i < 100000; i++) myglobal++;   /* NOT atomic! */
    vga_printf("A done. myglobal=%d\n", myglobal);
}
void thread_b(void) {
    for (int i = 0; i < 100000; i++) myglobal++;
    vga_printf("B done. myglobal=%d\n", myglobal);
}
/* Expected 200000 — without mutex, actual is somewhere 100000–200000 */
② Mutex — atomic XCHG + blocking waiter queue
static inline int atomic_xchg(volatile int *ptr, int val) {
    int old;
    __asm__ volatile("xchgl %0, %1"
        : "=r"(old), "+m"(*ptr) : "0"(val) : "memory");
    return old;
}

void mutex_lock(mutex_t *m) {
    while (atomic_xchg(&m->locked, 1) == 1) {
        m->waiters[m->nwaiters++] = current_tid;
        proc_table[current_tid].state = PROC_BLOCKED;
        scheduler_yield();
    }
    m->owner = current_tid;
}

void mutex_unlock(mutex_t *m) {
    m->owner = -1; m->locked = 0;
    if (m->nwaiters > 0)
        proc_table[m->waiters[--m->nwaiters]].state = PROC_READY;
}
③ Semaphore + producer-consumer validation
/* Three-semaphore bounded buffer — Lecture 11 §3 */
semaphore_t sem_empty, sem_full, sem_mutex;
int buffer[BUF_SIZE];

void producer(void) {
    for (;;) {
        int item = produce_item();
        sem_wait  (&sem_empty);  /* wait for empty slot */
        sem_wait  (&sem_mutex);  /* enter critical section */
        insert_item(buffer, item);
        sem_signal(&sem_mutex);  /* leave critical section */
        sem_signal(&sem_full);   /* signal full slot */
    }
}

Physical Memory Manager

Stage 3
① E820 memory map — detect RAM in Real Mode before PM switch
; boot.asm — call BEFORE LGDT/CR0 (still in Real Mode)
detect_memory:
    mov  di, 0x8004     ; write E820 entries starting at 0x8004
    xor  ebx, ebx        ; EBX=0 to begin
    xor  bp,  bp         ; entry count
.loop:
    mov  eax, 0xE820
    mov  ecx, 24         ; 24-byte entry
    mov  edx, 0x534D4150 ; 'SMAP' magic
    int  0x15
    jc   .done           ; CF set = end or error
    inc  bp  add  di, 24
    test ebx, ebx  jnz .loop
.done: mov [0x8000], bp  ; store count
② Bitmap allocator — 1 bit per 4 KB frame
static uint32_t bitmap[32*1024*1024/FRAME_SIZE/32]; /* 1 KB for 32 MB */

static inline void bitmap_set  (uint32_t f){bitmap[f/32] |=  (1u<<(f%32));}
static inline void bitmap_clear(uint32_t f){bitmap[f/32] &= ~(1u<<(f%32));}
static inline int  bitmap_test (uint32_t f){return(bitmap[f/32]>>(f%32))&1;}

uint32_t pmm_alloc_frame(void) {
    for (uint32_t i = 0; i < total_frames; i++) {
        if (!bitmap_test(i)) {
            bitmap_set(i); free_frames--;
            return i * FRAME_SIZE;  /* physical address */
        }
    }
    return 0;  /* OOM */
}
void pmm_free_frame(uint32_t phys) {
    bitmap_clear(phys/FRAME_SIZE); free_frames++;
}
